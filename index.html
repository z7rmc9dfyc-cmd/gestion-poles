function chargeurFichierPDF(event) { const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = async function(e) { try { const typedarray = new Uint8Array(e.target.result); const pdf = await pdfjsLib.getDocument(typedarray).promise; let texteComplet = ""; for (let i = 1; i <= pdf.numPages; i++) { const page = await pdf.getPage(i); const textContent = await page.getTextContent(); const pageText = textContent.items.map(item => item.str).join(" "); texteComplet += pageText + " "; } const corpsDeT = document.getElementById('corps-tableau-discours'); corpsDeT.innerHTML = ""; // Nettoyage des entêtes du PDF let cleanText = texteComplet.replace(/DATE THEME N° ORATEUR ASSEMBLEE PRESIDENT LECTEUR GROUPE qui reçoit l'orateur/gi, '').trim(); // RegEx pour découper proprement chaque ligne par sa date (format DD/MM/YYYY) const entryRegex = /(\d{2}\/\d{2}\/\d{4})(.*?)(?=\d{2}\/\d{2}\/\d{4}|PROGRAMME|$)/gs; let match; let ligneAjoutee = false; while ((match = entryRegex.exec(cleanText)) !== null) { const date = match[1]; const reste = match[2].trim(); // On cherche le numéro du thème (1 à 3 chiffres) qui sépare le Thème de l'Orateur const numMatch = reste.match(/(\d{1,3})\s+(.+)/); if (numMatch) { ligneAjoutee = true; const theme = reste.split(/\s\d{1,3}\s/)[0].trim(); const numero = numMatch[1]; const suite = numMatch[2].trim(); // Découpage des éléments restants const mots = suite.split(/\s+/); const orateur = `${mots[0] || ''} ${mots[1] || ''}`.trim(); const assemblee = mots[2] || ''; const president = `${mots[3] || ''} ${mots[4] || ''}`.trim(); const lecteur = `${mots[5] || ''} ${mots[6] || ''}`.trim(); const groupe = mots.slice(7).join(' ') || '--'; const tr = document.createElement('tr'); tr.innerHTML = ` ${date} ${theme} ${numero} ${orateur} ${assemblee} ${president} ${lecteur} ${groupe} `; corpsDeT.appendChild(tr); } } if (ligneAjoutee) { alert("Fichier PDF chargé et structuré avec succès !"); } else { alert("Le PDF a été lu, mais le format n'a pas pu être découpé."); } } catch (error) { console.error(error); alert("Erreur lors de la lecture du fichier PDF."); } }; reader.readAsArrayBuffer(file); }
